\section{Research Interests}
\begin{itemize}[leftmargin=*,topsep=6pt,itemsep=4pt,parsep=0pt]
\item \textbf{Robot learning and control:} Learning-based policies and motion planning for manipulation, locomotion, and whole-body control.
\item \textbf{3D perception and geometric representations:} Scene reconstruction, shape completion, and structured representations for collision-aware planning and interaction.
\item \textbf{Efficient physics simulation and learned dynamics:} Collision detection, contact modeling, and dynamics learning for fast simulation and policy training.
\item \textbf{Sim-to-real transfer and adaptation:} System identification, actuation modeling, and learning-based adaptive control for robust deployment across robots and operating conditions.
\end{itemize}
